cmake_minimum_required(VERSION 3.20)
project(ets2_lane_ingame LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/utf-8)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${ROOT}/vendor/minhook" minhook)
add_library(imgui STATIC
 "${ROOT}/vendor/imgui/imgui.cpp" "${ROOT}/vendor/imgui/imgui_draw.cpp"
 "${ROOT}/vendor/imgui/imgui_tables.cpp" "${ROOT}/vendor/imgui/imgui_widgets.cpp"
 "${ROOT}/vendor/imgui/backends/imgui_impl_dx11.cpp" "${ROOT}/vendor/imgui/backends/imgui_impl_win32.cpp")
target_include_directories(imgui PUBLIC "${ROOT}/vendor/imgui")
target_link_libraries(imgui PUBLIC d3d11 dxgi d3dcompiler imm32 dwmapi)
add_library(engine STATIC ../ingame_engine.cpp)
target_compile_options(engine PRIVATE /W4)
add_library(signals STATIC ../ingame_signals.cpp)
target_compile_options(signals PRIVATE /W4 /WX)
add_library(gps STATIC ../ingame_gps.cpp)
target_compile_options(gps PRIVATE /W4 /WX)
target_link_libraries(gps PUBLIC engine)
add_library(ets2_lane_ingame SHARED ../ingame_plugin.cpp ../ingame_renderer.cpp ../ingame_plugin.def)
target_include_directories(ets2_lane_ingame PRIVATE "${ROOT}/vendor/scs-sdk/include")
target_compile_options(ets2_lane_ingame PRIVATE /W4)
target_link_libraries(ets2_lane_ingame PRIVATE engine signals gps imgui minhook version)
add_executable(ingame_engine_test ../ingame_engine_test.cpp)
target_link_libraries(ingame_engine_test PRIVATE engine)
add_executable(ingame_signals_test ../ingame_signals_test.cpp)
target_link_libraries(ingame_signals_test PRIVATE signals)
add_executable(ingame_gps_test ../ingame_gps_test.cpp)
target_link_libraries(ingame_gps_test PRIVATE gps)
add_executable(ingame_render_host ../ingame_render_host.cpp)
target_include_directories(ingame_render_host PRIVATE "${ROOT}/vendor/scs-sdk/include")
target_link_libraries(ingame_render_host PRIVATE d3d11 dxgi)
enable_testing()
add_test(NAME navigation COMMAND ingame_engine_test "${ROOT}/data/ingame-map.bin")
add_test(NAME signals COMMAND ingame_signals_test)
add_test(NAME gps COMMAND ingame_gps_test)
# The render host is run explicitly: it creates a visible native DX11 window.
